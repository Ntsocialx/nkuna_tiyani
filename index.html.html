Student & Accommodation System Analysis

Detailed Data Flow Diagram (DFD) and system architecture analyzing student registration, property availability management, booking capture, bursary funding verification, and administrative reporting.

mysql_schema.sql 
-- Database Initialization
CREATE DATABASE BookstoreDB;
USE BookstoreDB;

-- Schema Construction
CREATE TABLE Authors (
    AuthorID INT PRIMARY KEY AUTO_INCREMENT,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL
);

CREATE TABLE Books (
    BookID INT PRIMARY KEY AUTO_INCREMENT,
    Title VARCHAR(150) NOT NULL,
    ISBN VARCHAR(20) UNIQUE NOT NULL,
    Price DECIMAL(10,2) NOT NULL,
    AuthorID INT,
    FOREIGN KEY (AuthorID) REFERENCES Authors(AuthorID) ON DELETE CASCADE
);

-- Queries & Relational Joins
SELECT b.Title, b.Price, CONCAT(a.FirstName, ' ', a.LastName) AS Author
FROM Books b
JOIN Authors a ON b.AuthorID = a.AuthorID
WHERE b.Price > 200.00;
